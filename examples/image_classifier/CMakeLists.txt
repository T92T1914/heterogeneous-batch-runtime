cmake_minimum_required(VERSION 3.24)
project(hbr_image_classifier LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
if(NOT HBR_ORT_INCLUDE OR NOT EXISTS "${HBR_ORT_INCLUDE}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set HBR_ORT_INCLUDE to the separately acquired, hash-verified ONNX Runtime 1.30 headers")
endif()
pybind11_add_module(_session session.cpp)
target_compile_features(_session PRIVATE cxx_std_20)
target_include_directories(_session PRIVATE "${HBR_ORT_INCLUDE}")
target_link_libraries(_session PRIVATE ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(_session PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(_session PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
install(TARGETS _session LIBRARY DESTINATION hbr_image_classifier RUNTIME DESTINATION hbr_image_classifier)
