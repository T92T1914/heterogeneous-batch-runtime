cmake_minimum_required(VERSION 3.24)
project(heterogeneous_batch_runtime VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(HBR_PYTHON "Build the narrow Python extension" OFF)
option(HBR_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(HBR_THREAD_SANITIZER "Enable separate host ThreadSanitizer coverage" OFF)
option(HBR_AVX2 "Compile runtime-dispatched AVX2 kernels on x86" ON)
option(HBR_CUDA "Build the optional CUDA backend and executed tests" OFF)
option(HBR_SYCL "Build the optional SYCL backend with a compiler that accepts -fsycl" OFF)
set(HBR_SYCL_TARGETS "nvidia_gpu_sm_89" CACHE STRING "Device targets for -fsycl-targets when HBR_SYCL is ON")
set(HBR_SYCL_FLAGS "" CACHE STRING "Extra SYCL compile and link options as a CMake list, such as --cuda-path=<toolkit>. Use SHELL: for an option that must repeat")
find_package(Threads REQUIRED)
if(HBR_THREAD_SANITIZER)
  if(HBR_SANITIZERS OR HBR_CUDA OR HBR_SYCL OR HBR_PYTHON OR MSVC)
    message(FATAL_ERROR "ThreadSanitizer requires a separate GCC/Clang host-only build")
  endif()
endif()
add_library(hbr src/kernels.cpp src/runtime.cpp)
target_compile_features(hbr PUBLIC cxx_std_20)
target_include_directories(hbr PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hbr PUBLIC Threads::Threads)
set_target_properties(hbr PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(hbr PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(hbr PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(HBR_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "(AMD64|amd64|x86_64|x64)")
  target_sources(hbr PRIVATE src/avx2.cpp)
  target_compile_definitions(hbr PRIVATE HBR_HAS_AVX2=1)
  if(MSVC)
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  else()
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
  endif()
endif()
if(HBR_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use GCC or Clang for the ASan/UBSan contract")
  endif()
  target_compile_options(hbr PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hbr PUBLIC -fsanitize=address,undefined)
endif()
if(HBR_THREAD_SANITIZER)
  target_compile_options(hbr PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(hbr PUBLIC -fsanitize=thread)
endif()
if(BUILD_TESTING)
  add_executable(hbr_tests tests/native.cpp)
  target_link_libraries(hbr_tests PRIVATE hbr)
  add_test(NAME native_contract COMMAND hbr_tests)
  set_tests_properties(native_contract PROPERTIES TIMEOUT 60)
endif()
add_executable(hbr_bench tools/benchmark.cpp)
target_link_libraries(hbr_bench PRIVATE hbr)
if(HBR_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(hbr_cuda src/cuda.cu)
  target_compile_features(hbr_cuda PUBLIC cxx_std_20)
  target_include_directories(hbr_cuda PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  set_target_properties(hbr_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  set_target_properties(hbr_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(hbr_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(hbr_cuda PUBLIC CUDA::cudart_static)
  install(TARGETS hbr_cuda EXPORT hbrTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  add_executable(hbr_cuda_bench tools/cuda_benchmark.cpp)
  target_link_libraries(hbr_cuda_bench PRIVATE hbr hbr_cuda)
  add_executable(hbr_cuda_reuse_bench tools/cuda_reuse_benchmark.cpp)
  target_link_libraries(hbr_cuda_reuse_bench PRIVATE hbr hbr_cuda)
  add_executable(hbr_cuda_pinned_bench tools/cuda_pinned_benchmark.cpp)
  target_link_libraries(hbr_cuda_pinned_bench PRIVATE hbr hbr_cuda)
  if(BUILD_TESTING)
    add_executable(hbr_cuda_tests tests/cuda.cpp)
    target_link_libraries(hbr_cuda_tests PRIVATE hbr hbr_cuda)
    add_test(NAME cuda_contract COMMAND hbr_cuda_tests)
    set_tests_properties(cuda_contract PROPERTIES TIMEOUT 120)
    add_executable(hbr_cuda_reuse_tests tests/cuda_reuse.cpp)
    target_link_libraries(hbr_cuda_reuse_tests PRIVATE hbr_cuda Threads::Threads)
    add_test(NAME cuda_reuse_contract COMMAND hbr_cuda_reuse_tests)
    add_executable(hbr_cuda_reuse_faults tests/cuda_reuse_faults.cpp src/cuda.cu)
    target_compile_features(hbr_cuda_reuse_faults PRIVATE cxx_std_20)
    set_target_properties(hbr_cuda_reuse_faults PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
    target_compile_definitions(hbr_cuda_reuse_faults PRIVATE HBR_TEST_CUDA_FAULTS=1)
    target_compile_options(hbr_cuda_reuse_faults PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
    target_include_directories(hbr_cuda_reuse_faults PRIVATE include)
    target_link_libraries(hbr_cuda_reuse_faults PRIVATE CUDA::cudart_static Threads::Threads)
    add_test(NAME cuda_reuse_faults COMMAND hbr_cuda_reuse_faults)
    set_tests_properties(cuda_reuse_contract cuda_reuse_faults PROPERTIES TIMEOUT 120)
  endif()
endif()
if(HBR_SYCL)
  # Built and tested only with DPC++ on Linux. The library is shared so that it carries
  # its own device image and an application links it without a SYCL compiler.
  if(WIN32)
    message(FATAL_ERROR "HBR_SYCL is built and tested only on Linux with DPC++")
  endif()
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-fsycl")
  set(CMAKE_REQUIRED_LINK_OPTIONS -fsycl)
  check_cxx_source_compiles("#include <sycl/sycl.hpp>\nint main() { return sycl::device::get_devices().size() > 1024; }" HBR_SYCL_COMPILER)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  if(NOT HBR_SYCL_COMPILER)
    message(FATAL_ERROR "HBR_SYCL needs a C++ compiler that accepts -fsycl and provides <sycl/sycl.hpp>, such as the DPC++ clang++. Set CMAKE_CXX_COMPILER to it.")
  endif()
  set(HBR_SYCL_OPTIONS -fsycl "-fsycl-targets=${HBR_SYCL_TARGETS}" ${HBR_SYCL_FLAGS})
  add_library(hbr_sycl SHARED src/sycl.cpp)
  target_compile_features(hbr_sycl PUBLIC cxx_std_20)
  target_include_directories(hbr_sycl PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_compile_options(hbr_sycl PRIVATE ${HBR_SYCL_OPTIONS} -Wall -Wextra -ffp-contract=off)
  target_link_options(hbr_sycl PRIVATE ${HBR_SYCL_OPTIONS})
  install(TARGETS hbr_sycl EXPORT hbrTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  if(HBR_CUDA)
    # The timing comparison calls both backends in one process. Configure it with the DPC++
    # clang++ as the C++ compiler and set CMAKE_CUDA_HOST_COMPILER for nvcc. It opens the
    # driver's NVML library at run time to read the card's load, so it links only dlopen.
    add_executable(hbr_sycl_cuda_bench tools/sycl_cuda_benchmark.cpp)
    target_link_libraries(hbr_sycl_cuda_bench PRIVATE hbr hbr_cuda hbr_sycl ${CMAKE_DL_LIBS})
  endif()
  if(BUILD_TESTING)
    # Tests whose names start with sycl_ need a SYCL GPU. host_sycl_selection hides every
    # GPU and expects a visible rejection, so it runs on any host.
    add_executable(hbr_sycl_tests tests/sycl.cpp)
    target_link_libraries(hbr_sycl_tests PRIVATE hbr_sycl)
    add_test(NAME sycl_device COMMAND hbr_sycl_tests)
    add_test(NAME host_sycl_selection COMMAND hbr_sycl_tests --expect-no-gpu)
    set_tests_properties(host_sycl_selection PROPERTIES ENVIRONMENT "ONEAPI_DEVICE_SELECTOR=*:cpu")
    add_executable(hbr_sycl_histogram_tests tests/sycl_histogram.cpp)
    target_link_libraries(hbr_sycl_histogram_tests PRIVATE hbr hbr_sycl)
    add_test(NAME sycl_histogram COMMAND hbr_sycl_histogram_tests)
    add_executable(hbr_sycl_reduction_tests tests/sycl_reduction.cpp)
    target_link_libraries(hbr_sycl_reduction_tests PRIVATE hbr hbr_sycl)
    add_test(NAME sycl_reduction COMMAND hbr_sycl_reduction_tests)
    add_executable(hbr_sycl_stencil_tests tests/sycl_stencil.cpp)
    target_link_libraries(hbr_sycl_stencil_tests PRIVATE hbr hbr_sycl)
    add_test(NAME sycl_stencil COMMAND hbr_sycl_stencil_tests)
    # The fault build compiles the library source again with test checkpoints, so it needs -fsycl.
    add_executable(hbr_sycl_faults tests/sycl_faults.cpp src/sycl.cpp)
    target_compile_definitions(hbr_sycl_faults PRIVATE HBR_TEST_SYCL_FAULTS=1)
    target_compile_options(hbr_sycl_faults PRIVATE ${HBR_SYCL_OPTIONS} -ffp-contract=off)
    target_link_options(hbr_sycl_faults PRIVATE ${HBR_SYCL_OPTIONS})
    target_link_libraries(hbr_sycl_faults PRIVATE hbr)
    add_test(NAME sycl_faults COMMAND hbr_sycl_faults)
    set_tests_properties(sycl_device host_sycl_selection sycl_histogram sycl_reduction sycl_stencil sycl_faults PROPERTIES TIMEOUT 120)
  endif()
endif()
if(HBR_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native src/bindings.cpp)
  target_link_libraries(_native PRIVATE hbr)
  if(HBR_CUDA)
    target_compile_definitions(_native PRIVATE HBR_PYTHON_CUDA=1)
    target_link_libraries(_native PRIVATE hbr_cuda)
  endif()
  install(TARGETS _native LIBRARY DESTINATION heterogeneous_batch_runtime RUNTIME DESTINATION heterogeneous_batch_runtime)
  if(BUILD_TESTING)
    pybind11_add_module(_array_boundary_test tests/python_arrays.cpp)
    target_link_libraries(_array_boundary_test PRIVATE hbr)
    pybind11_add_module(_host_cuda_binding_test src/bindings.cpp tests/cuda_binding_stub.cpp)
    target_compile_definitions(_host_cuda_binding_test PRIVATE HBR_PYTHON_CUDA=1 HBR_BINDINGS_MODULE=_host_cuda_binding_test)
    target_include_directories(_host_cuda_binding_test PRIVATE tests/cuda_stub)
    target_link_libraries(_host_cuda_binding_test PRIVATE hbr)
    add_test(NAME python_array_boundary COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/python_array_boundary.py"
      "$<TARGET_FILE:_array_boundary_test>" "$<TARGET_FILE:_host_cuda_binding_test>")
    set_tests_properties(python_array_boundary PROPERTIES TIMEOUT 60)
  endif()
endif()
install(TARGETS hbr EXPORT hbrTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT hbrTargets NAMESPACE hbr:: DESTINATION lib/cmake/hbr)

configure_package_config_file(cmake/hbrConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hbrConfig.cmake" INSTALL_DESTINATION lib/cmake/hbr)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hbrConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hbrConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hbrConfigVersion.cmake" DESTINATION lib/cmake/hbr)
