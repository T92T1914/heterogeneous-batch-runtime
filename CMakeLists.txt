cmake_minimum_required(VERSION 3.24)
project(heterogeneous_batch_runtime VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(HBR_PYTHON "Build the narrow Python extension" OFF)
option(HBR_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(HBR_THREAD_SANITIZER "Enable separate host ThreadSanitizer coverage" OFF)
option(HBR_AVX2 "Compile runtime-dispatched AVX2 kernels on x86" ON)
option(HBR_CUDA "Build the optional CUDA backend and executed tests" OFF)
find_package(Threads REQUIRED)
if(HBR_THREAD_SANITIZER)
  if(HBR_SANITIZERS OR HBR_CUDA OR HBR_PYTHON OR MSVC)
    message(FATAL_ERROR "ThreadSanitizer requires a separate GCC/Clang host-only build")
  endif()
endif()
add_library(hbr src/kernels.cpp src/runtime.cpp)
target_compile_features(hbr PUBLIC cxx_std_20)
target_include_directories(hbr PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hbr PUBLIC Threads::Threads)
set_target_properties(hbr PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(hbr PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(hbr PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(HBR_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "(AMD64|amd64|x86_64|x64)")
  target_sources(hbr PRIVATE src/avx2.cpp)
  target_compile_definitions(hbr PRIVATE HBR_HAS_AVX2=1)
  if(MSVC)
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  else()
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
  endif()
endif()
if(HBR_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use GCC or Clang for the ASan/UBSan contract")
  endif()
  target_compile_options(hbr PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hbr PUBLIC -fsanitize=address,undefined)
endif()
if(HBR_THREAD_SANITIZER)
  target_compile_options(hbr PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(hbr PUBLIC -fsanitize=thread)
endif()
if(BUILD_TESTING)
  add_executable(hbr_tests tests/native.cpp)
  target_link_libraries(hbr_tests PRIVATE hbr)
  add_test(NAME native_contract COMMAND hbr_tests)
  set_tests_properties(native_contract PROPERTIES TIMEOUT 60)
endif()
add_executable(hbr_bench tools/benchmark.cpp)
target_link_libraries(hbr_bench PRIVATE hbr)
if(HBR_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(hbr_cuda src/cuda.cu)
  target_compile_features(hbr_cuda PUBLIC cxx_std_20)
  target_include_directories(hbr_cuda PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  set_target_properties(hbr_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
  set_target_properties(hbr_cuda PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(hbr_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(hbr_cuda PUBLIC CUDA::cudart_static)
  install(TARGETS hbr_cuda EXPORT hbrTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  add_executable(hbr_cuda_bench tools/cuda_benchmark.cpp)
  target_link_libraries(hbr_cuda_bench PRIVATE hbr hbr_cuda)
  add_executable(hbr_cuda_reuse_bench tools/cuda_reuse_benchmark.cpp)
  target_link_libraries(hbr_cuda_reuse_bench PRIVATE hbr hbr_cuda)
  if(BUILD_TESTING)
    add_executable(hbr_cuda_tests tests/cuda.cpp)
    target_link_libraries(hbr_cuda_tests PRIVATE hbr hbr_cuda)
    add_test(NAME cuda_contract COMMAND hbr_cuda_tests)
    set_tests_properties(cuda_contract PROPERTIES TIMEOUT 120)
    add_executable(hbr_cuda_reuse_tests tests/cuda_reuse.cpp)
    target_link_libraries(hbr_cuda_reuse_tests PRIVATE hbr_cuda Threads::Threads)
    add_test(NAME cuda_reuse_contract COMMAND hbr_cuda_reuse_tests)
    add_executable(hbr_cuda_reuse_faults tests/cuda_reuse_faults.cpp src/cuda.cu)
    target_compile_features(hbr_cuda_reuse_faults PRIVATE cxx_std_20)
    set_target_properties(hbr_cuda_reuse_faults PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
    target_compile_definitions(hbr_cuda_reuse_faults PRIVATE HBR_TEST_CUDA_FAULTS=1)
    target_compile_options(hbr_cuda_reuse_faults PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
    target_include_directories(hbr_cuda_reuse_faults PRIVATE include)
    target_link_libraries(hbr_cuda_reuse_faults PRIVATE CUDA::cudart_static Threads::Threads)
    add_test(NAME cuda_reuse_faults COMMAND hbr_cuda_reuse_faults)
    set_tests_properties(cuda_reuse_contract cuda_reuse_faults PROPERTIES TIMEOUT 120)
  endif()
endif()
if(HBR_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native src/bindings.cpp)
  target_link_libraries(_native PRIVATE hbr)
  if(HBR_CUDA)
    target_compile_definitions(_native PRIVATE HBR_PYTHON_CUDA=1)
    target_link_libraries(_native PRIVATE hbr_cuda)
  endif()
  install(TARGETS _native LIBRARY DESTINATION heterogeneous_batch_runtime RUNTIME DESTINATION heterogeneous_batch_runtime)
endif()
install(TARGETS hbr EXPORT hbrTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT hbrTargets NAMESPACE hbr:: DESTINATION lib/cmake/hbr)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/hbrConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/hbrConfig.cmake" INSTALL_DESTINATION lib/cmake/hbr)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/hbrConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hbrConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/hbrConfigVersion.cmake" DESTINATION lib/cmake/hbr)
