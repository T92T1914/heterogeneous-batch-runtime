cmake_minimum_required(VERSION 3.24)
project(hbr_package_consumer LANGUAGES CXX)
find_package(hbr 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE hbr::hbr)
if(TARGET hbr::hbr_cuda)
  target_link_libraries(consumer PRIVATE hbr::hbr_cuda)
  target_compile_definitions(consumer PRIVATE CHECK_CUDA=1)
endif()
