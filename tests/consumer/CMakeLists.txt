cmake_minimum_required(VERSION 3.24)
project(hbr_package_consumer LANGUAGES CXX)
option(HBR_CONSUMER_CUDA "Require and execute the installed CUDA interface" OFF)
option(HBR_CONSUMER_SYCL "Require and execute the installed SYCL interface" OFF)
set(hbr_components cpu)
if(HBR_CONSUMER_CUDA)
  list(APPEND hbr_components cuda)
endif()
if(HBR_CONSUMER_SYCL)
  list(APPEND hbr_components sycl)
endif()
find_package(hbr 0.1 CONFIG REQUIRED COMPONENTS ${hbr_components})
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE hbr::hbr)
if(HBR_CONSUMER_CUDA)
  target_link_libraries(consumer PRIVATE hbr::hbr_cuda)
  target_compile_definitions(consumer PRIVATE CHECK_CUDA=1)
endif()
if(HBR_CONSUMER_SYCL)
  target_link_libraries(consumer PRIVATE hbr::hbr_sycl)
  target_compile_definitions(consumer PRIVATE CHECK_SYCL=1)
endif()
