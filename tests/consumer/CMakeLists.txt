cmake_minimum_required(VERSION 3.24)
project(hbr_package_consumer LANGUAGES CXX)
option(HBR_CONSUMER_CUDA "Require and execute the installed CUDA interface" OFF)
if(HBR_CONSUMER_CUDA)
  find_package(hbr 0.1 CONFIG REQUIRED COMPONENTS cpu cuda)
else()
  find_package(hbr 0.1 CONFIG REQUIRED COMPONENTS cpu)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE hbr::hbr)
if(HBR_CONSUMER_CUDA)
  target_link_libraries(consumer PRIVATE hbr::hbr_cuda)
  target_compile_definitions(consumer PRIVATE CHECK_CUDA=1)
endif()
